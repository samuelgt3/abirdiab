import { describe, it, expect } from 'vitest';
import { bookingEmail, esc, joinEmail, safeLink, zonedToUtc } from '../../supabase/functions/_shared/emails.ts';

const session = {
  ref: 'AD-3C5E15', kind: 'session', item_title: '1:1 Session', customer_name: 'Layla Hassan', email: 'l@example.com', lang: 'en',
  status: 'awaiting_payment', session_date: '2026-11-20', session_time: '10:00', session_timezone: 'Asia/Dubai', amount_usd: 90, delivery_method: 'email',
};

describe('booking email', () => {
  it('thanks by first name, shows the time and reference; a session link comes straight after payment', () => {
    const m = bookingEmail(session, 'https://www.abirdiab.com/');
    expect(m.subject).toContain('AD-3C5E15');
    expect(m.html).toContain('Hi Layla,');
    expect(m.html).toContain('10:00 (Asia/Dubai)');
    expect(m.html).toContain('as soon as your payment is received');
    expect(m.html).not.toContain('2 hours before');
    expect(m.html).toContain('https://www.abirdiab.com/account?tab=sessions');
  });
  it('gives a paid session its own joining link straight away', () => {
    const m = bookingEmail({ ...session, status: 'paid', join_url: 'https://meet.jit.si/AbirDiab-abc123' }, 'https://x.com');
    expect(m.html).toContain('https://meet.jit.si/AbirDiab-abc123');
    expect(m.html).not.toContain('2 hours before');
    expect(m.text).toContain('https://meet.jit.si/AbirDiab-abc123');
    expect(bookingEmail({ ...session, status: 'paid', join_url: 'javascript:alert(1)' }, 'https://x.com').html).not.toContain('javascript:');
  });
  it('never links to a pay page (everything is paid at checkout); only an unpaid order says it is confirmed on payment', () => {
    expect(bookingEmail(session, 'https://x.com').html).not.toContain('/pay');
    expect(bookingEmail(session, 'https://x.com').html).toContain('confirmed as soon as your payment is received');
    expect(bookingEmail({ ...session, status: 'paid' }, 'https://x.com').html).not.toContain('confirmed as soon as');
  });
  it('is written in Arabic, right-to-left, when the customer used Arabic', () => {
    const m = bookingEmail({ ...session, lang: 'ar' }, 'https://x.com');
    expect(m.html).toContain('dir="rtl"');
    expect(m.html).toContain('شكرًا');
  });
  it('mentions WhatsApp delivery instead of email when chosen', () => {
    const m = bookingEmail({ ...session, delivery_method: 'whatsapp' }, 'https://x.com');
    expect(m.html).not.toContain('emailed to you');
  });
  it('escapes anything a customer typed', () => {
    const m = bookingEmail({ ...session, customer_name: '<script>alert(1)</script>' }, 'https://x.com');
    expect(m.html).not.toContain('<script>');
    expect(esc('"a" & <b>')).toBe('&quot;a&quot; &amp; &lt;b&gt;');
  });
});

describe('joining-link email', () => {
  it('contains the link, button and time', () => {
    const m = joinEmail(session, 'https://zoom.us/j/123', 'https://x.com');
    expect(m.html).toContain('https://zoom.us/j/123');
    expect(m.subject).toContain('joining link');
  });
  it('only accepts https links', () => {
    expect(safeLink('https://zoom.us/j/1')).toBe('https://zoom.us/j/1');
    expect(safeLink('javascript:alert(1)')).toBeNull();
    expect(safeLink('http://zoom.us')).toBeNull();
    expect(safeLink('')).toBeNull();
  });
});

describe('zonedToUtc', () => {
  it('converts a Dubai wall-clock time (UTC+4) to UTC', () => {
    expect(zonedToUtc('2026-11-20', '10:00', 'Asia/Dubai').toISOString()).toBe('2026-11-20T06:00:00.000Z');
  });
});

import { giftEmail } from '../../supabase/functions/_shared/emails.ts';

describe('gift email', () => {
  const g = { token: 'abc123', sender_name: 'Sara', recipient_name: 'Layla Hassan', message: 'Enjoy <3', lang: 'en' };
  it('greets the receiver, names the sender and links to the claim page', () => {
    const m = giftEmail(g, 'Intro to Reiki', 'https://www.abirdiab.com/');
    expect(m.subject).toContain('Sara');
    expect(m.html).toContain('Hi Layla,');
    expect(m.html).toContain('https://www.abirdiab.com/gift/abc123');
    expect(m.html).toContain('Enjoy &lt;3');
  });
  it('is right-to-left in Arabic', () => {
    expect(giftEmail({ ...g, lang: 'ar' }, 'x', 'https://x.com').html).toContain('dir="rtl"');
  });
});

describe('purchase email for every kind', () => {
  const base = { ref: 'AD-1A2B3C4D5', customer_name: 'Mona Ali', email: 'mona@example.com', lang: 'en', status: 'paid', item_title: 'Miraculous Monday', amount_usd: 9 };
  it('workshop: says the joining link comes 2 hours before and opens My workshops', () => {
    const m = bookingEmail({ ...base, kind: 'workshop', item_title: 'Year-End Release Circle', delivery_method: 'email' }, 'https://x.com', '2026-12-12T16:00:00Z');
    expect(m.subject).toBe('Your booking: Year-End Release Circle (AD-1A2B3C4D5)');
    expect(m.html).toContain('2 hours before');
    expect(m.html).toContain('/account?tab=workshops');
    expect(m.html).toContain('mona@example.com');
  });
  it('course: says it is unlocked and points to My learning, using the email as the login', () => {
    const m = bookingEmail({ ...base, kind: 'course' }, 'https://x.com');
    expect(m.subject).toBe('Your order: Miraculous Monday (AD-1A2B3C4D5)');
    expect(m.html).toContain('course is unlocked');
    expect(m.html).toContain('/account?tab=learning');
    expect(m.html).toContain('sign in or create your password using mona@example.com');
    expect(m.html).not.toContain('2 hours');
  });
  it('gift card: tells the buyer the link is in My gifts; gift to a person: says we email them', () => {
    const card = bookingEmail({ ...base, kind: 'gift' }, 'https://x.com');
    expect(card.html).toContain('Thank you for your gift');
    expect(card.html).toContain('My gifts');
    expect(card.html).toContain('/account?tab=gifts');
    const direct = bookingEmail({ ...base, kind: 'gift', gift_to_name: 'Sara', gift_to_email: 'sara@example.com' }, 'https://x.com');
    expect(direct.html).toContain('We are emailing Sara their gift');
  });
  it('is written in Arabic for Arabic customers, for courses and gifts too', () => {
    expect(bookingEmail({ ...base, kind: 'course', lang: 'ar' }, 'https://x.com').html).toContain('رحلة التعلم');
    expect(bookingEmail({ ...base, kind: 'gift', lang: 'ar' }, 'https://x.com').html).toContain('الهدايا');
  });
});
